\hypertarget{appendix-v-the-python-history-and-pep-timeline-1989-2026}{%
\chapter{Appendix V: The Python History and PEP Timeline
(1989-2026)}\label{appendix-v-the-python-history-and-pep-timeline-1989-2026}}

This appendix provides a detailed chronological timeline of Python's
development, mapping major releases to the PEPs that defined them.

\hypertarget{v.1-the-pre-release-years-1989-1991}{%
\subsection{V.1 The Pre-Release Years
(1989-1991)}\label{v.1-the-pre-release-years-1989-1991}}

\begin{itemize}
\tightlist
\item
  \textbf{1989 (Dec)}: Guido van Rossum starts Python as a Christmas
  project.
\item
  \textbf{1991 (Feb)}: Python 0.9.0 posted to alt.sources.
\end{itemize}

\hypertarget{v.2-the-1.x-era-formalization-1994-2000}{%
\subsection{V.2 The 1.x Era: Formalization
(1994-2000)}\label{v.2-the-1.x-era-formalization-1994-2000}}

\begin{itemize}
\tightlist
\item
  \textbf{1994 (Jan)}: Python 1.0 (Functional programming features).
\item
  \textbf{1995}: Python 1.2.
\item
  \textbf{1997}: Python 1.5 (Standard library expansion).
\end{itemize}

\hypertarget{v.3-the-2.x-era-the-modern-foundation-2000-2020}{%
\subsection{V.3 The 2.x Era: The Modern Foundation
(2000-2020)}\label{v.3-the-2.x-era-the-modern-foundation-2000-2020}}

\begin{itemize}
\tightlist
\item
  \textbf{2000 (Oct)}: Python 2.0 (Comprehensions, GC, Unicode support).
\item
  \textbf{2001}: Python 2.1 (Nested scopes,
  \passthrough{\lstinline!\_\_future\_\_!}).
\item
  \textbf{2002}: Python 2.2 (Type-class unification).
\item
  \textbf{2003}: Python 2.3 (C3 linearization).
\item
  \textbf{2004}: Python 2.4 (Decorators).
\item
  \textbf{2006}: Python 2.5 (\passthrough{\lstinline!with!} statement).
\item
  \textbf{2008}: Python 2.6 (Transition release).
\item
  \textbf{2010}: Python 2.7 (The long-term support release).
\item
  \textbf{2020 (Jan)}: Python 2.7 officially EOL (End of Life).
\end{itemize}

\hypertarget{v.4-the-3.x-era-the-great-schism-and-growth-2008-present}{%
\subsection{V.4 The 3.x Era: The Great Schism and Growth
(2008-Present)}\label{v.4-the-3.x-era-the-great-schism-and-growth-2008-present}}

\begin{itemize}
\tightlist
\item
  \textbf{2008 (Dec)}: Python 3.0 (Unicode by default,
  \passthrough{\lstinline!print!} is a function).
\item
  \textbf{2009}: Python 3.1 (Ordered dicts).
\item
  \textbf{2011}: Python 3.2 (GIL improvements).
\item
  \textbf{2012}: Python 3.3 (\passthrough{\lstinline!yield from!},
  namespace packages).
\item
  \textbf{2014}: Python 3.4 (\passthrough{\lstinline!asyncio!},
  \passthrough{\lstinline!pathlib!}, \passthrough{\lstinline!enum!}).
\item
  \textbf{2015}: Python 3.5
  (\passthrough{\lstinline!async!}/\passthrough{\lstinline!await!}).
\item
  \textbf{2016}: Python 3.6 (\passthrough{\lstinline!f-strings!},
  variable annotations).
\item
  \textbf{2018}: Python 3.7 (\passthrough{\lstinline!dataclasses!},
  context variables).
\item
  \textbf{2019}: Python 3.8 (\passthrough{\lstinline!walrus operator!}).
\item
  \textbf{2020}: Python 3.9 (\passthrough{\lstinline!PEG parser!}, dict
  merge).
\item
  \textbf{2021}: Python 3.10
  (\passthrough{\lstinline!structural pattern matching!}).
\item
  \textbf{2022}: Python 3.11
  (\passthrough{\lstinline!specializing interpreter!}).
\item
  \textbf{2023}: Python 3.12
  (\passthrough{\lstinline!per-interpreter GIL!},
  \passthrough{\lstinline!generics!}).
\item
  \textbf{2024}: Python 3.13 (\passthrough{\lstinline!free-threading!},
  \passthrough{\lstinline!JIT!}).
\item
  \textbf{2025}: Python 3.14 (Planned enhancements to Tier-2 optimizer).
\item
  \textbf{2026}: Python 3.15 (Expected focus on WASM and mobile
  support).
\end{itemize}
