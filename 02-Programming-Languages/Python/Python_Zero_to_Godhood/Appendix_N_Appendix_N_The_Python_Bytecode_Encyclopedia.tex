\chapter{Appendix N: The Python Bytecode
Encyclopedia}\label{appendix-n-the-python-bytecode-encyclopedia}}

This appendix provides an exhaustive reference for the CPython 3.13
instruction set. For each opcode, we provide its numerical value, its
stack transition, and a technical description of its C-level
implementation.

\hypertarget{n.1-stack-notation}{%
\subsection{N.1 Stack Notation}\label{n.1-stack-notation}}

\begin{itemize}
\tightlist
\item
  \passthrough{\lstinline!TOS!}: Top of Stack.
\item
  \passthrough{\lstinline!TOS1!}: Second item on stack.
\item
  \passthrough{\lstinline!TOS2!}: Third item on stack.
\item
  \passthrough{\lstinline!NULL!}: A sentinel value.
\end{itemize}

\toprule
Opcode & Transition & Description \\
\midrule
\endhead
\passthrough{\lstinline!LOAD\_CONST!} &
\passthrough{\lstinline!( -> const)!} & Pushes a constant from the code

\passthrough{\lstinline!LOAD\_FAST!} &
\passthrough{\lstinline!( -> val)!} & Pushes a local variable onto the

\passthrough{\lstinline!STORE\_FAST!} &
\passthrough{\lstinline!(val -> )!} & Pops the top of the stack and
stores it in a local variable. \\
\passthrough{\lstinline!LOAD\_GLOBAL!} &
\passthrough{\lstinline!( -> val)!} & Pushes a global or builtin name
onto the stack. Uses the \passthrough{\lstinline!co\_names!} tuple and
performs a hash table lookup. \\
\passthrough{\lstinline!STORE\_GLOBAL!} &
\passthrough{\lstinline!(val -> )!} & Stores the top of the stack in the
global namespace. \\
\passthrough{\lstinline!DELETE\_GLOBAL!} &
\passthrough{\lstinline!( -> )!} & Deletes a global name. \\
\bottomrule
\end{longtable}

\toprule
Opcode & Transition & Description \\
\midrule
\endhead
\passthrough{\lstinline!BINARY\_OP!} &
\passthrough{\lstinline!(TOS1, TOS -> result)!} & General opcode for

\passthrough{\lstinline!UNARY\_NEGATIVE!} &

\passthrough{\lstinline!(TOS -> not TOS)!} & Performs logical NOT. \\
\passthrough{\lstinline!UNARY\_INVERT!} &
\passthrough{\lstinline!(TOS -> \~TOS)!} & Performs bitwise
inversion. \\
\bottomrule

\hypertarget{n.5-control-flow-and-function-calls}{%
\subsection{N.5 Control Flow and Function
Calls}\label{n.5-control-flow-and-function-calls}}

\begin{longtable}[]{@{}
  >{\raggedright\arraybackslash}p{(\columnwidth - 4\tabcolsep) * \real{0.33}}
  >{\raggedright\arraybackslash}p{(\columnwidth - 4\tabcolsep) * \real{0.33}}
  >{\raggedright\arraybackslash}p{(\columnwidth - 4\tabcolsep) * \real{0.33}}@{}}
\toprule
Opcode & Transition & Description \\
\midrule
\endhead
\passthrough{\lstinline!JUMP\_FORWARD!} &
\passthrough{\lstinline!( -> )!} & Increments the instruction
pointer. \\
\passthrough{\lstinline!POP\_JUMP\_IF\_FALSE!} &
\passthrough{\lstinline!(bool -> )!} & Jumps if the top of the stack is
false. \\
\passthrough{\lstinline!CALL!} &
\passthrough{\lstinline!(func, args -> result)!} & Calls a function. In
3.11+, this is the ``mega-opcode'' that handles all function calls,
including those with keyword arguments. \\
\passthrough{\lstinline!RETURN\_VALUE!} &
\passthrough{\lstinline!(val -> )!} & Returns the top of the stack to
the caller's frame. \\
\passthrough{\lstinline!RAISE\_VARARGS!} &
\passthrough{\lstinline!(val -> )!} & Raises an exception. \\
\bottomrule
\end{longtable}
