\hypertarget{p.2-the-small-object-allocator-workflow}{%
\subsection{P.2 The Small Object Allocator
Workflow}\label{p.2-the-small-object-allocator-workflow}}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{Request}: Python requests 32 bytes for a small string.
\item
  \textbf{Size Class}: \passthrough{\lstinline!PyMalloc!} identifies
  this as Size Class 3.
\item
  \textbf{Pool Check}: It checks the \passthrough{\lstinline!usedpools!}
  array for Size Class 3.
\item
  \textbf{Block Return}: It returns a pointer to the next free block in
  the pool.
\item
  \textbf{Alignment}: Blocks are always 8-byte aligned to ensure
  hardware efficiency (Chapter 70).
\end{enumerate}
