\chapter{Appendix W: Python in Mobile and Embedded

Python is increasingly used beyond the server and desktop, in
environments with extreme resource constraints.

\hypertarget{w.1-python-on-android-and-ios-beeware-and-kivy}{%
\subsection{W.1 Python on Android and iOS: BeeWare and
Kivy}\label{w.1-python-on-android-and-ios-beeware-and-kivy}}

\begin{itemize}
\tightlist
\item
  \textbf{BeeWare}: A suite of tools to write native Python apps for
  mobile and desktop.
\item
  \textbf{Briefcase}: Packages Python code as native Android/iOS
  installers.
\end{itemize}

\hypertarget{w.2-micropython-and-circuitpython}{%
\subsection{W.2 MicroPython and
CircuitPython}\label{w.2-micropython-and-circuitpython}}

\begin{itemize}
\tightlist
\item
  \textbf{MicroPython}: A lean and efficient implementation of Python 3
  that includes a small subset of the Python standard library and is
  optimized to run on microcontrollers.
\item
  \textbf{CircuitPython}: A fork of MicroPython designed for educational
  purposes, supported by Adafruit.
\item
  \textbf{The VIP (Virtual Integer Program)}: How MicroPython handles
  memory without a standard OS heap.
\end{itemize}
