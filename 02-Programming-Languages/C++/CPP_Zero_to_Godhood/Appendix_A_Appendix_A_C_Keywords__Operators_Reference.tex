\hypertarget{essential-keywords-non-exhaustive}{%
\subsubsection{Essential Keywords
(Non-Exhaustive)}\label{essential-keywords-non-exhaustive}}

\begin{itemize}
\tightlist
\item
  \textbf{alignas / alignof}: Memory alignment queries and
  specifications.
\item
  \textbf{asm}: Inline assembly block.
\item
  \textbf{auto}: Type deduction (C++11).
\item
  \textbf{const / volatile}: cv-qualifiers for type safety and hardware
  access.
\item
  \textbf{constexpr / consteval / constinit}: Compile-time constant
  specifications.
\item
  \textbf{decltype}: Inspect declared type of an entity.
\item
  \textbf{explicit}: Prevent implicit conversions in constructors.
\item
  \textbf{export}: Module interface export (C++20).
\item
  \textbf{friend}: Allow access to private members.
\item
  \textbf{inline}: Suggest inlining to compiler; allow definition in
  header.
\item
  \textbf{mutable}: Allow modification of member in const object.
\item
  \textbf{noexcept}: Specifier for functions that don't throw.
\item
  \textbf{nullptr}: Null pointer literal (C++11).
\item
  \textbf{operator}: Overload operators.
\item
  \textbf{requires}: Constraint clause for Concepts (C++20).
\item
  \textbf{static\_assert}: Compile-time assertion.
\item
  \textbf{template}: Define generic classes/functions.
\item
  \textbf{thread\_local}: Storage duration specifier.
\item
  \textbf{typeid}: Runtime type identification (RTTI).
\item
  \textbf{typename}: Declare a type parameter in templates.
\item
  \textbf{virtual}: Declare virtual function for polymorphism.
\end{itemize}
