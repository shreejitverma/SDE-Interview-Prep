\hypertarget{eigen-linear-algebra}{%
\subsection{23.1 Eigen (Linear Algebra)}\label{eigen-linear-algebra}}

Template-heavy library that avoids temporaries using Expression
Templates.

\begin{lstlisting}[language={C++}]
#include <Eigen/Dense>
using Eigen::MatrixXd;

void solve_system() {
    MatrixXd A(3, 3);
    A << 1, 2, 3,
         4, 5, 6,
         7, 8, 10;
    
    Eigen::VectorXd b(3);
    b << 3, 3, 4;
    
    Eigen::VectorXd x = A.colPivHouseholderQr().solve(b);
}
\end{lstlisting}

\hypertarget{cuda-gpu-programming}{%
\subsection{23.2 CUDA (GPU Programming)}\label{cuda-gpu-programming}}

Running C++ directly on NVIDIA GPUs.

\begin{lstlisting}[language={C++}]
// Kernel (runs on GPU)
__global__ void vectorAdd(float* A, float* B, float* C, int N) {
    int i = blockDim.x * blockIdx.x + threadIdx.x;
    if (i < N) C[i] = A[i] + B[i];
}

// Host (runs on CPU)
void launch_kernel(float* d_A, float* d_B, float* d_C, int N) {
    int threadsPerBlock = 256;
    int blocksPerGrid = (N + threadsPerBlock - 1) / threadsPerBlock;
    vectorAdd<<<blocksPerGrid, threadsPerBlock>>>(d_A, d_B, d_C, N);
}
\end{lstlisting}
