\hypertarget{stdgenerator}{%
\subsection{\texorpdfstring{1.
\texttt{std::generator}}{1. std::generator}}\label{stdgenerator}}

Standardized generator coroutine. Supports
\passthrough{\lstinline!co\_yield!} and recursive usage.

\hypertarget{basic-generator}{%
\subsubsection{1.1 Basic Generator}\label{basic-generator}}

\begin{lstlisting}[language={C++}]
#include <generator>
#include <ranges>

std::generator<int> seq(int start) {
    while (true) {
        co_yield start++;
    }
}

int main() {
    for (int i : seq(0) | std::views::take(5)) {
        std::println("{}", i);
    }
}
\end{lstlisting}

\hypertarget{recursive-generator}{%
\subsubsection{1.2 Recursive Generator}\label{recursive-generator}}

\passthrough{\lstinline!std::generator!} supports
\passthrough{\lstinline!co\_yield ranges::elements\_of(...)!} to yield
values from a sub-generator or range efficiently.

\hypertarget{stdstacktrace}{%
\subsection{\texorpdfstring{2.
\texttt{std::stacktrace}}{2. std::stacktrace}}\label{stdstacktrace}}

Obtain the call stack at runtime. Useful for logging and debugging.

\begin{lstlisting}[language={C++}]
#include <stacktrace>
#include <print>

void boom() {
    auto trace = std::stacktrace::current();
    std::println("{}", std::to_string(trace));
}
\end{lstlisting}

\textbf{Note:} Requires compiler/linker flags (e.g.,
\passthrough{\lstinline!-lstdc++\_libbacktrace!} on GCC).
