Data Parallelism: Processing 8 or 16 numbers in a single CPU cycle.

\hypertarget{simd-basics-registers}{%
\subsection{32.1 SIMD Basics \& Registers}\label{simd-basics-registers}}

\begin{itemize}
\tightlist
\item
  \textbf{SSE}: 128-bit (4 floats). XMM registers.
\item
  \textbf{AVX2}: 256-bit (8 floats). YMM registers.
\item
  \textbf{AVX-512}: 512-bit (16 floats). ZMM registers.
\end{itemize}

\hypertarget{intrinsics-example-vector-addition}{%
\subsection{32.2 Intrinsics Example (Vector
Addition)}\label{intrinsics-example-vector-addition}}

Using \passthrough{\lstinline!<immintrin.h>!}.

\begin{lstlisting}[language={C++}]
#include <immintrin.h>

void add_avx2(float* a, float* b, float* c, int N) {
    // Process 8 floats at a time
    for (int i = 0; i < N; i += 8) {
        // Load
        __m256 va = _mm256_loadu_ps(&a[i]);
        __m256 vb = _mm256_loadu_ps(&b[i]);
        
        // Operation
        __m256 vc = _mm256_add_ps(va, vb);
        
        // Store
        _mm256_storeu_ps(&c[i], vc);
    }
}
\end{lstlisting}

\begin{itemize}
\tightlist
\item
  \passthrough{\lstinline!\_mm256\_loadu\_ps!}: Load Unaligned Packed
  Single-precision.
\item
  \passthrough{\lstinline!\_mm256\_add\_ps!}: Add packed singles.
\end{itemize}

\hypertarget{measurable-outcome}{%
\subsection{32.3 Measurable Outcome}\label{measurable-outcome}}

\begin{itemize}
\tightlist
\item
  \textbf{Objective}: Convert a scalar loop to AVX2.

  bandwidth permitting).
\end{itemize}
