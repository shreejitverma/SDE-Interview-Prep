\chapter{Appendix D: Common C++ Traps \&
Pitfalls}\label{appendix-d-common-c-traps-pitfalls}}

\hypertarget{i.-general-syntax-traps}{%
\subsection{I. General \& Syntax Traps}\label{i.-general-syntax-traps}}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{Most Vexing Parse}

  \begin{itemize}
  \tightlist
  \item
    \emph{Issue}: \passthrough{\lstinline!MyClass obj();!} declares a
    function returning \passthrough{\lstinline!MyClass!}, not a
    default-constructed object.
  \item
    \emph{Fix}: Use brace initialization:
    \passthrough{\lstinline!MyClass obj\{\};!}.
  \end{itemize}
\item
  \textbf{The ``dangling else'' Problem}

  \begin{itemize}
  \tightlist
  \item
    \emph{Issue}: Nested \passthrough{\lstinline!if-else!} without
    braces can associate \passthrough{\lstinline!else!} with the wrong
    \passthrough{\lstinline!if!}.
  \item
    \emph{Fix}: Always use braces \passthrough{\lstinline!\{\}!} for
    control structures.
  \end{itemize}
\item
  \textbf{Integer Division}

  \begin{itemize}
  \tightlist
  \item
    \emph{Issue}: \passthrough{\lstinline!1/2!} results in
    \passthrough{\lstinline!0!} (integer), not
    \passthrough{\lstinline!0.5!}.
  \item
    \emph{Fix}: Cast one operand to float/double:
    \passthrough{\lstinline!1.0/2!} or
    \passthrough{\lstinline!static\_cast<double>(1)/2!}.
  \end{itemize}
\item
  \textbf{Loop Variable Type Mismatch}

  \begin{itemize}
  \tightlist
  \item
    \emph{Issue}:
    \passthrough{\lstinline!for (unsigned i = v.size() - 1; i >= 0; --i)!}
    causes an infinite loop because \passthrough{\lstinline!unsigned!}
    is never negative.
  \item
    \emph{Fix}: Use \passthrough{\lstinline!int!} (and cast size) or
    standard iterators/ranges.
  \end{itemize}
\item
  \textbf{Shadowing Variables}

  \begin{itemize}
  \tightlist
  \item
    \emph{Issue}: Declaring a local variable with the same name as a
    member or outer variable hides the outer one.
  \item
    \emph{Fix}: Enable compiler warnings
    (\passthrough{\lstinline!-Wshadow!}) and use
    \passthrough{\lstinline!this->member!} if necessary.
  \end{itemize}
\end{enumerate}

\hypertarget{ii.-pointers-references-memory}{%
\subsection{II. Pointers, References \&
Memory}\label{ii.-pointers-references-memory}}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\setcounter{enumi}{5}
\tightlist
\item
  \textbf{Object Slicing}

  \begin{itemize}
  \tightlist
  \item
    \emph{Issue}: Assigning a \passthrough{\lstinline!Derived!} object
    to a \passthrough{\lstinline!Base!} value slices off the derived
    part.
  \item
    \emph{Fix}: Use pointers \passthrough{\lstinline!Base*!} or
    references \passthrough{\lstinline!Base\&!} for polymorphism.
  \end{itemize}
\item
  \textbf{Dangling References}

  \begin{itemize}
  \tightlist
  \item
    \emph{Issue}: Returning a reference to a local stack variable.
  \item
    \emph{Fix}: Return by value or use smart pointers/dynamic
    allocation.
  \end{itemize}
\item
  \textbf{Iterator Invalidation}

  \begin{itemize}
  \tightlist
  \item
    \emph{Issue}: Adding elements to a
    \passthrough{\lstinline!std::vector!} may reallocate memory,
    invalidating all pointers/iterators to elements.
  \item
    \emph{Fix}: Don't cache iterators across mutating operations; use
    \passthrough{\lstinline!reserve()!} if possible.
  \end{itemize}
\item
  \textbf{\passthrough{\lstinline!delete!} vs
  \passthrough{\lstinline!delete[]!}}

  \begin{itemize}
  \tightlist
  \item
    \emph{Issue}: Mismatching \passthrough{\lstinline!new!} with
    \passthrough{\lstinline!delete[]!} or
    \passthrough{\lstinline!new[]!} with
    \passthrough{\lstinline!delete!} causes undefined behavior.
  \item
    \emph{Fix}: Use \passthrough{\lstinline!std::vector!} or
    \passthrough{\lstinline!std::unique\_ptr!} instead of manual
    management.
  \end{itemize}
\item
  \textbf{Use-After-Move}

  \begin{itemize}
  \tightlist
  \item
    \emph{Issue}: Accessing an object after
    \passthrough{\lstinline!std::move()!} (except for
    reassignment/destruction).
  \item
    \emph{Fix}: Treat moved-from objects as empty; do not read their
    state.
  \end{itemize}
\end{enumerate}

\hypertarget{iii.-classes-oop}{%
\subsection{III. Classes \& OOP}\label{iii.-classes-oop}}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\setcounter{enumi}{10}
\tightlist
\item
  \textbf{Virtual Destructor Missing}

  \begin{itemize}
  \tightlist
  \item
    \emph{Issue}: Deleting a derived class via a base pointer when the
    base destructor is not \passthrough{\lstinline!virtual!} leaks
    derived resources.
  \item
    \emph{Fix}: Always mark base class destructors
    \passthrough{\lstinline!virtual!} (or
    \passthrough{\lstinline!protected!} if non-polymorphic).
  \end{itemize}
\item
  \textbf{Calling Virtual Functions in Constructor/Destructor}

  \begin{itemize}
  \tightlist
  \item
    \emph{Issue}: Calls the \emph{base} class version, not the derived
    one, because the derived part isn't initialized/is already
    destroyed.
  \item
    \emph{Fix}: Use two-phase initialization or factory methods.
  \end{itemize}
\item
  \textbf{Copy Constructor/Assignment Missing}

  \begin{itemize}
  \tightlist
  \item
    \emph{Issue}: Classes managing raw pointers will default to shallow
    copy (double free error).
  \item
    \emph{Fix}: Follow the \textbf{Rule of Three/Five/Zero}.
  \end{itemize}
\item
  \textbf{Initialization Order}

  \begin{itemize}
  \tightlist
  \item
    \emph{Issue}: Members are initialized in \emph{declaration order},
    not initializer list order.
  \item
    \emph{Fix}: Keep initializer list order identical to member
    declaration order to avoid warnings.
  \end{itemize}
\end{enumerate}

\hypertarget{iv.-concurrency}{%
\subsection{IV. Concurrency}\label{iv.-concurrency}}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\setcounter{enumi}{14}
\tightlist
\item
  \textbf{Data Races}

  \begin{itemize}
  \tightlist
  \item
    \emph{Issue}: Multiple threads accessing shared memory without
    synchronization (at least one writer).
  \item
    \emph{Fix}: Use \passthrough{\lstinline!std::mutex!},
    \passthrough{\lstinline!std::atomic!}, or
    \passthrough{\lstinline!std::shared\_mutex!}.
  \end{itemize}
\item
  \textbf{Deadlocks}

  \begin{itemize}
  \tightlist
  \item
    \emph{Issue}: Two threads waiting on each other's locks.
  \item
    \emph{Fix}: Acquire locks in a consistent global order; use
    \passthrough{\lstinline!std::scoped\_lock!} (C++17) to lock multiple
    mutexes safely.
  \end{itemize}
\item
  \textbf{False Sharing}

  \begin{itemize}
  \tightlist
  \item
    \emph{Issue}: Independent atomic variables on the same cache line
    degrade performance due to cache coherency protocols.
  \item
    \emph{Fix}: Use
    \passthrough{\lstinline!alignas(hardware\_destructive\_interference\_size)!}
    to pad variables.
  \end{itemize}
\end{enumerate}

\hypertarget{v.-modern-c-macros}{%
\subsection{V. Modern C++ \& Macros}\label{v.-modern-c-macros}}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\setcounter{enumi}{17}
\tightlist
\item
  \textbf{\passthrough{\lstinline!std::vector<bool>!} Weirdness}

  \begin{itemize}
  \tightlist
  \item
    \emph{Issue}: It's a template specialization (bitfield), not a
    vector of bools. Returns a proxy object, not
    \passthrough{\lstinline!bool\&!}.
  \item
    \emph{Fix}: Use \passthrough{\lstinline!std::deque<bool>!} or
    \passthrough{\lstinline!std::vector<char>!} if you need real
    references.
  \end{itemize}
\item
  \textbf{Auto Type Deduction}

  \begin{itemize}
  \tightlist
  \item
    \emph{Issue}: \passthrough{\lstinline!auto!} drops references and
    \passthrough{\lstinline!const!}.
  \item
    \emph{Fix}: Use \passthrough{\lstinline!auto\&!} or
    \passthrough{\lstinline!const auto\&!} explicitly when needed.
  \end{itemize}
\item
  \textbf{Macro Side Effects}

  \begin{itemize}
  \tightlist
  \item
    \emph{Issue}:
    \passthrough{\lstinline!\#define MAX(a,b) ((a) > (b) ? (a) : (b))!}
    evaluates arguments twice. \passthrough{\lstinline!MAX(x++, y)!}
    increments \passthrough{\lstinline!x!} twice.
  \item
    \emph{Fix}: Use \passthrough{\lstinline!inline!} functions or
    templates instead of macros.
  \end{itemize}
\item
  \textbf{Static Initialization Order Fiasco}

  \begin{itemize}
  \tightlist
  \item
    \emph{Issue}: Global objects in different files have undefined
    initialization order.
  \item
    \emph{Fix}: Use the ``Construct On First Use'' idiom (Meyers
    Singleton).
  \end{itemize}
\end{enumerate}
