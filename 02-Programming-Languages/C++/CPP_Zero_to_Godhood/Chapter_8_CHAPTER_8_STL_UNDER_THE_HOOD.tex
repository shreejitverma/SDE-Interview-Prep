\hypertarget{the-truth-about-stdvector}{%
\subsubsection{1. The Truth About
std::vector}\label{the-truth-about-stdvector}}

\passthrough{\lstinline!std::vector!} is a dynamic array. It guarantees
contiguous memory.

\begin{itemize}
\tightlist
\item
  \textbf{Layout}: Three pointers: \passthrough{\lstinline!start!},
  \passthrough{\lstinline!finish!},
  \passthrough{\lstinline!end\_of\_storage!}.

  \begin{itemize}
  \tightlist
  \item
    \passthrough{\lstinline!start!}: Points to first element.
  \item
    \passthrough{\lstinline!finish!}: Points to one-past-the-last active
    element (size).
  \item
    \passthrough{\lstinline!end\_of\_storage!}: Points to end of
    allocated buffer (capacity).
  \end{itemize}
\item
  \textbf{Growth Strategy}: Geometric growth.

  \begin{itemize}
  \tightlist
  \item
    When \passthrough{\lstinline!size() == capacity()!}, a new buffer is
    allocated (usually 2x or 1.5x larger).
  \item
    \textbf{Elements are Copied} to the new buffer (C++98 does not have
    Move semantics).
  \item
    Old buffer is deleted.
  \item
    \emph{Cost}: Amortized O(1) push\_back, but worst-case O(N).
  \end{itemize}
\item
  \textbf{Iterator Invalidation}:

  \begin{itemize}
  \tightlist
  \item
    \textbf{Reallocation}: Invalidates ALL iterators, pointers, and
    references.
  \item
    \textbf{Insertion/Erasure}: Invalidates iterators at and after the
    point of operation.
  \end{itemize}
\end{itemize}

\hypertarget{the-stddeque-implementation}{%
\subsubsection{2. The std::deque
Implementation}\label{the-stddeque-implementation}}

\passthrough{\lstinline!std::deque!} (Double-Ended Queue) is NOT a
contiguous array.

\begin{itemize}
\tightlist
\item
  \textbf{Layout}: A ``Map'' (dynamic array) of pointers to fixed-size
  ``Chunks'' (blocks).

  \begin{itemize}
  \tightlist
  \item
    Iterators are smart pointers that know how to jump between chunks.
  \end{itemize}
\item
  \textbf{Performance}:

  \begin{itemize}
  \tightlist
  \item
    O(1) random access (double dereference).
  \item
    O(1) push/pop at BOTH ends (no full reallocation needed, just add a
    new chunk).
  \end{itemize}
\item
  \textbf{Cache Locality}: Worse than vector, better than list.
\end{itemize}

\hypertarget{why-stdlist-is-almost-always-wrong}{%
\subsubsection{3. Why std::list is (Almost) Always
Wrong}\label{why-stdlist-is-almost-always-wrong}}

\passthrough{\lstinline!std::list!} is a Doubly Linked List.

\begin{itemize}
\tightlist
\item
  \textbf{Layout}: Nodes allocated individually on the heap.

  \begin{itemize}
  \tightlist
  \item
    \passthrough{\lstinline!struct Node \{ T val; Node* prev; Node* next; \}!}
  \end{itemize}
\item
  \textbf{The Cache Problem}: Nodes are scattered in memory. Traversing
  a list causes constant \textbf{Cache Misses}.
\item
  \textbf{Benchmark}: Iterating a \passthrough{\lstinline!vector!} is
  orders of magnitude faster than a \passthrough{\lstinline!list!}, even
  for large types, due to prefetching.
\item
  \textbf{Use Case}: Only when you need \textbf{Reference Stability}
  (insertions never invalidate references to other elements) or frequent
  splicing.
\end{itemize}

\hypertarget{associative-containers-mapset}{%
\subsubsection{4. Associative Containers
(Map/Set)}\label{associative-containers-mapset}}

\passthrough{\lstinline!std::map!}, \passthrough{\lstinline!std::set!},
\passthrough{\lstinline!std::multimap!},
\passthrough{\lstinline!std::multiset!}.

\begin{itemize}
\tightlist
\item
  \textbf{Implementation}: Balanced Binary Search Tree (usually
  \textbf{Red-Black Tree}).
\item
  \textbf{Node Layout}:
  \passthrough{\lstinline!struct Node \{ T val; Node* left; Node* right; Node* parent; Color color; \}!}
\item
  \textbf{Complexity}: O(log N) for insert, lookup, delete.
\item
  \textbf{Overhead}: 3 pointers + enum per element (heavy memory
  overhead).
\end{itemize}

\hypertarget{iterator-invalidation-cheat-sheet}{%
\subsubsection{5. Iterator Invalidation Cheat
Sheet}\label{iterator-invalidation-cheat-sheet}}

\begin{longtable}[]{@{}lll@{}}
\toprule
Container & Operation & Invalidates \\
\midrule
\endhead
\textbf{Vector} & Capacity Change & \textbf{ALL} \\
\textbf{Vector} & Insert/Erase & Current \& After \\
\textbf{Deque} & Insert/Erase (ends) & Iterators only (Refs valid!) \\
\textbf{Deque} & Insert/Erase (middle) & \textbf{ALL} \\
\textbf{List} & Insert/Erase & Only deleted element \\
\textbf{Map/Set} & Insert/Erase & Only deleted element \\
\bottomrule
\end{longtable}
