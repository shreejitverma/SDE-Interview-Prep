\hypertarget{hardware-sympathy}{%
\subsubsection{1. Hardware Sympathy}\label{hardware-sympathy}}

\begin{itemize}
\tightlist
\item
  \textbf{\passthrough{\lstinline!std::unreachable!}}: Tells the
  optimizer that a specific branch of code can never be reached. If it
  is reached, it is Undefined Behavior. This allows the compiler to
  strip out safety checks.
  \passthrough{\lstinline!cpp     enum class State \{ A, B \};     void f(State s) \{         if (s == State::A) do\_a();         else if (s == State::B) do\_b();         else std::unreachable(); // Compiler optimizes knowing this is impossible     \}!}
\item
  \textbf{\passthrough{\lstinline!std::byteswap!}}: Highly optimized
  byte reversal (endianness swap), mapping directly to compiler
  intrinsics like \passthrough{\lstinline!bswap!}.
\end{itemize}

\hypertarget{core-utilities}{%
\subsubsection{2. Core Utilities}\label{core-utilities}}

\begin{itemize}
\tightlist
\item
  \textbf{\passthrough{\lstinline!std::to\_underlying!}}: A safe, clean
  way to extract the numeric value of an
  \passthrough{\lstinline!enum class!}.
  \passthrough{\lstinline!cpp     enum class Flags : uint8\_t \{ Read = 1, Write = 2 \};     auto val = std::to\_underlying(Flags::Write); // uint8\_t 2!}
\item
  \textbf{\passthrough{\lstinline!std::move\_only\_function!}}: A
  lightweight version of \passthrough{\lstinline!std::function!} that
  can hold non-copyable callables (like lambdas capturing
  \passthrough{\lstinline!std::unique\_ptr!}). It has significantly less
  overhead.
\item
  \textbf{\passthrough{\lstinline!std::string::contains!}}: Finally, a
  readable way to check for substrings without comparing against
  \passthrough{\lstinline!std::string::npos!}.
  \passthrough{\lstinline!cpp     if (str.contains("error")) \{ /* ... */ \}!}
\end{itemize}
