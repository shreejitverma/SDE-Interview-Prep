\hypertarget{hardware-memory}{%
\subsubsection{1. Hardware \& Memory}\label{hardware-memory}}

\begin{itemize}
\tightlist
\item
  \textbf{std::unreachable()}: Marks code that should never be reached.
  Gives the compiler optimization permission and causes UB if reached.
  \passthrough{\lstinline!cpp     default: std::unreachable();!}
\item
  \textbf{std::byteswap}: Reverses the byte order of an integral value;
  useful for endianness conversion.
  \passthrough{\lstinline!cpp     uint32\_t be = std::byteswap(0x01020304u);!}
\item
  \textbf{std::out\_ptr / std::inout\_ptr}: Helpers for passing smart
  pointers to legacy C APIs that expect \passthrough{\lstinline!T**!}
  output parameters.
  \passthrough{\lstinline!cpp     legacy\_init(std::out\_ptr(my\_unique\_ptr));!}
\item
  \textbf{std::spanstream}: A string stream that operates on a fixed
  \passthrough{\lstinline!std::span<char>!} buffer rather than
  allocating heap memory (faster than
  \passthrough{\lstinline!stringstream!}).
  \passthrough{\lstinline!cpp     std::spanstream ss\{buf\}; ss << 42;!}
\end{itemize}

\hypertarget{functional-utilities}{%
\subsubsection{2. Functional Utilities}\label{functional-utilities}}

\begin{itemize}
\tightlist
\item
  \textbf{std::move\_only\_function}: Like
  \passthrough{\lstinline!std::function!} but only move-constructible;
  supports move-only callables (lambdas capturing
  \passthrough{\lstinline!unique\_ptr!}) and avoids unnecessary copies.
\item
  \textbf{std::to\_underlying}: Converts an enum to its underlying
  integer type without a \passthrough{\lstinline!static\_cast!}.
\item
  \textbf{string::contains}: \passthrough{\lstinline!std::string!} and
  \passthrough{\lstinline!std::string\_view!} gain
  \passthrough{\lstinline!.contains()!} to check for substring presence.
\end{itemize}

\hypertarget{math-constexpr-upgrades}{%
\subsubsection{3. Math \& Constexpr
Upgrades}\label{math-constexpr-upgrades}}

\begin{itemize}
\tightlist
\item
  \textbf{Fixed-width floating-point types}:
  \passthrough{\lstinline!<stdfloat>!} introduces
  \passthrough{\lstinline!std::float16\_t!},
  \passthrough{\lstinline!std::float32\_t!},
  \passthrough{\lstinline!std::float64\_t!}, and
  \passthrough{\lstinline!std::bfloat16\_t!} (if supported by platform).
\item
  \textbf{constexpr upgrades}: \passthrough{\lstinline!std::optional!},
  \passthrough{\lstinline!std::variant!},
  \passthrough{\lstinline!std::unique\_ptr!}, and many
  \passthrough{\lstinline!<cmath>!} functions (e.g.,
  \passthrough{\lstinline!abs!}, \passthrough{\lstinline!ceil!}) are now
  fully \passthrough{\lstinline!constexpr!}.
\end{itemize}
