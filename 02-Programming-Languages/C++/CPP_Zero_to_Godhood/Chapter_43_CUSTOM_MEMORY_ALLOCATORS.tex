\hypertarget{custom-memory-allocators}{%
\chapter{CUSTOM MEMORY ALLOCATORS}\label{custom-memory-allocators}}

\passthrough{\lstinline!malloc!} and \passthrough{\lstinline!new!} are
general-purpose and slow (locks, fragmentation). Real-time systems use
custom allocators.

\hypertarget{linear-allocator-arena}{%
\subsection{33.1 Linear Allocator
(Arena)}\label{linear-allocator-arena}}

The absolute fastest allocator. O(1). Zero overhead.

\begin{lstlisting}[language={C++}]
class LinearAllocator {
    char* start;
    char* current;
    size_t size;
public:
    LinearAllocator(size_t s) : size(s) {
        start = new char[s];
        current = start;
    }
    
    void* allocate(size_t n) {
        if (current + n > start + size) return nullptr;
        void* ptr = current;
        current += n;
        return ptr;
    }
    
    void reset() { current = start; } // Free ALL at once
};
\end{lstlisting}

\begin{itemize}
\tightlist
\item
  \textbf{Use Case}: Per-frame game memory, Request-scoped web server
  memory.
\end{itemize}

\hypertarget{pool-allocator}{%
\subsection{33.2 Pool Allocator}\label{pool-allocator}}

Fixed-size blocks. No external fragmentation. O(1) malloc/free.

\begin{lstlisting}[language={C++}]
struct Chunk { Chunk* next; };

class PoolAllocator {
    Chunk* head = nullptr;
public:
    void* allocate() {
        if (!head) return ::operator new(sizeof(Chunk)); // Or expand pool
        Chunk* ptr = head;
        head = head->next;
        return ptr;
    }
    
    void deallocate(void* ptr) {
        Chunk* chunk = static_cast<Chunk*>(ptr);
        chunk->next = head;
        head = chunk;
    }
};
\end{lstlisting}
