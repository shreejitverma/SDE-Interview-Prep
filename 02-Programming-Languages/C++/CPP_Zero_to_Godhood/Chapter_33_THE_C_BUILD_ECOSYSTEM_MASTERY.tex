\hypertarget{vcpkg-manifest-mode}{%
\subsubsection{vcpkg (Manifest Mode)}\label{vcpkg-manifest-mode}}

Create \passthrough{\lstinline!vcpkg.json!} in your root:

\begin{lstlisting}
{
  "name": "my-app",
  "version": "1.0.0",
  "dependencies": [
    "fmt",
    "nlohmann-json"
  ]
}
\end{lstlisting}

CMake integration:

\begin{lstlisting}[language=bash]
cmake -B build -DCMAKE_TOOLCHAIN_FILE=.../vcpkg.cmake
\end{lstlisting}

\hypertarget{addresssanitizer-asan}{%
\subsubsection{AddressSanitizer (ASan)}\label{addresssanitizer-asan}}

Detects out-of-bounds, use-after-free.
\passthrough{\lstinline!clang++ -fsanitize=address -g main.cpp!}

\textbf{Example: Use-After-Free}

\begin{lstlisting}[language={C++}]
int* p = new int(5);
delete p;
*p = 10; // ASan catches this instantly!
\end{lstlisting}

\hypertarget{threadsanitizer-tsan}{%
\subsubsection{ThreadSanitizer (TSan)}\label{threadsanitizer-tsan}}

Detects data races.
\passthrough{\lstinline!clang++ -fsanitize=thread -g main.cpp!}

\textbf{Example: Data Race}

\begin{lstlisting}[language={C++}]
int counter = 0;
std::thread t1([&]{ counter++; });
std::thread t2([&]{ counter++; }); // TSan catches this race
t1.join(); t2.join();
\end{lstlisting}

\hypertarget{undefinedbehaviorsanitizer-ubsan}{%
\subsubsection{UndefinedBehaviorSanitizer
(UBSan)}\label{undefinedbehaviorsanitizer-ubsan}}

Detects overflow, null dereference, alignment issues.
\passthrough{\lstinline!clang++ -fsanitize=undefined -g main.cpp!}
