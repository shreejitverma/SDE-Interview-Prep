\section{Appendix F: The C++ Standard Evolution

\hypertarget{versioned-changelog}{%
\subsubsection{1. Versioned Changelog}\label{versioned-changelog}}

\hypertarget{c98-isoiec-148821998---the-foundation}{%
\paragraph{\texorpdfstring{\textbf{C++98 (ISO/IEC 14882:1998)} -
\emph{The
Foundation}}{C++98 (ISO/IEC 14882:1998) - The Foundation}}\label{c98-isoiec-148821998---the-foundation}}

\textbf{Released:} 1998 * \textbf{Core:} Templates, Exceptions,
Namespaces, \passthrough{\lstinline!bool!} type,
\passthrough{\lstinline!cast!} operators
(\passthrough{\lstinline!static\_cast!}, etc.),
\passthrough{\lstinline!mutable!}, \passthrough{\lstinline!explicit!}. *
\textbf{STL:} Containers (\passthrough{\lstinline!vector!},
\passthrough{\lstinline!list!}, \passthrough{\lstinline!map!},
\passthrough{\lstinline!set!}, \passthrough{\lstinline!deque!}),
Algorithms (\passthrough{\lstinline!sort!},
\passthrough{\lstinline!find!}, \passthrough{\lstinline!transform!}),
Iterators, Strings (\passthrough{\lstinline!std::string!}), I/O Streams
(\passthrough{\lstinline!iostream!}). * \textbf{Memory:}
\passthrough{\lstinline!std::auto\_ptr!} (Deprecated in C++11).

\hypertarget{c03-isoiec-148822003---the-bug-fix}{%
\paragraph{\texorpdfstring{\textbf{C++03 (ISO/IEC 14882:2003)} -
\emph{The Bug
Fix}}{C++03 (ISO/IEC 14882:2003) - The Bug Fix}}\label{c03-isoiec-148822003---the-bug-fix}}

\textbf{Released:} 2003 * \textbf{Focus:} Defect Report (DR) fixes for
C++98 to ensure consistency across compilers. * \textbf{Features:} Value
initialization \passthrough{\lstinline!T()!}, fixes to
\passthrough{\lstinline!std::vector!} contiguous memory guarantee.

\hypertarget{c11-isoiec-148822011---the-modern-revolution}{%
\paragraph{\texorpdfstring{\textbf{C++11 (ISO/IEC 14882:2011)} -
\emph{The Modern
Revolution}}{C++11 (ISO/IEC 14882:2011) - The Modern Revolution}}\label{c11-isoiec-148822011---the-modern-revolution}}

\textbf{Released:} September 2011 * \textbf{Language:}
\passthrough{\lstinline!auto!}, \passthrough{\lstinline!nullptr!},
Range-based \passthrough{\lstinline!for!}, Lambda expressions, Rvalue
references (\passthrough{\lstinline!\&\&!}) \& Move semantics, Variadic
templates, \passthrough{\lstinline!constexpr!} (limited),
\passthrough{\lstinline!decltype!}, Uniform initialization
\passthrough{\lstinline!\{\}!},
\passthrough{\lstinline!static\_assert!},
\passthrough{\lstinline!override!}, \passthrough{\lstinline!final!},
\passthrough{\lstinline!enum class!}. * \textbf{Concurrency:}
\passthrough{\lstinline!std::thread!},
\passthrough{\lstinline!std::mutex!},
\passthrough{\lstinline!std::atomic!},
\passthrough{\lstinline!std::future!},
\passthrough{\lstinline!std::async!}. * \textbf{Library:} Smart pointers
(\passthrough{\lstinline!unique\_ptr!},
\passthrough{\lstinline!shared\_ptr!},
\passthrough{\lstinline!weak\_ptr!}),
\passthrough{\lstinline!std::array!},
\passthrough{\lstinline!std::tuple!},
\passthrough{\lstinline!std::unordered\_map/set!},
\passthrough{\lstinline!std::regex!},
\passthrough{\lstinline!std::chrono!}.

\hypertarget{c14-isoiec-148822014---the-refinement}{%
\paragraph{\texorpdfstring{\textbf{C++14 (ISO/IEC 14882:2014)} -
\emph{The
Refinement}}{C++14 (ISO/IEC 14882:2014) - The Refinement}}\label{c14-isoiec-148822014---the-refinement}}

\textbf{Released:} December 2014 * \textbf{Language:} Generic lambdas
(\passthrough{\lstinline!auto!} params), Relaxed
\passthrough{\lstinline!constexpr!} (loops/variables allowed), Binary
literals (\passthrough{\lstinline!0b1010!}), Digit separators
(\passthrough{\lstinline!1'000!}), Variable templates, Return type
deduction. * \textbf{Library:}
\passthrough{\lstinline!std::make\_unique!},
\passthrough{\lstinline!std::shared\_timed\_mutex!},
\passthrough{\lstinline!std::integer\_sequence!},
\passthrough{\lstinline!std::exchange!},
\passthrough{\lstinline!std::quoted!}.

\hypertarget{c17-isoiec-148822017---the-major-update}{%
\paragraph{\texorpdfstring{\textbf{C++17 (ISO/IEC 14882:2017)} -
\emph{The Major
Update}}{C++17 (ISO/IEC 14882:2017) - The Major Update}}\label{c17-isoiec-148822017---the-major-update}}

\textbf{Released:} December 2017 * \textbf{Language:} Structured
bindings \passthrough{\lstinline!auto [x,y] = p;!},
\passthrough{\lstinline!if constexpr!}, Fold expressions
\passthrough{\lstinline!(... + args)!}, Class Template Argument
Deduction (CTAD), Inline variables,
\passthrough{\lstinline!\_\_has\_include!}. * \textbf{Library:}
\passthrough{\lstinline!std::filesystem!},
\passthrough{\lstinline!std::optional!},
\passthrough{\lstinline!std::variant!},
\passthrough{\lstinline!std::any!},
\passthrough{\lstinline!std::string\_view!}, Parallel Algorithms
(\passthrough{\lstinline!std::execution::par!}),
\passthrough{\lstinline!std::invoke!},
\passthrough{\lstinline!std::byte!}, \passthrough{\lstinline!std::pmr!}
(Polymorphic Memory Resources).

\hypertarget{c20-isoiec-148822020---the-gigantic-leap}{%
\paragraph{\texorpdfstring{\textbf{C++20 (ISO/IEC 14882:2020)} -
\emph{The Gigantic
Leap}}{C++20 (ISO/IEC 14882:2020) - The Gigantic Leap}}\label{c20-isoiec-148822020---the-gigantic-leap}}

\textbf{Released:} December 2020 * \textbf{Language:} Concepts
(Constraints), Modules (\passthrough{\lstinline!import/export!}),
Coroutines (\passthrough{\lstinline!co\_await!}), Three-way comparison
(\passthrough{\lstinline!<=>!}), Designated initializers
\passthrough{\lstinline!\{.x=1\}!}, \passthrough{\lstinline!consteval!}
(Immediate functions), \passthrough{\lstinline!constinit!}, Range-based
for with init. * \textbf{Library:} Ranges
(\passthrough{\lstinline!std::ranges!}),
\passthrough{\lstinline!std::span!},
\passthrough{\lstinline!std::format!},
\passthrough{\lstinline!std::jthread!},
\passthrough{\lstinline!std::stop\_token!},
\passthrough{\lstinline!std::barrier!},
\passthrough{\lstinline!std::latch!},
\passthrough{\lstinline!std::semaphore!},
\passthrough{\lstinline!std::bit\_cast!},
\passthrough{\lstinline!std::source\_location!}, Calendars \& Timezones.

\hypertarget{c23-isoiec-148822023---the-completion}{%
\paragraph{\texorpdfstring{\textbf{C++23 (ISO/IEC 14882:2023)} -
\emph{The
Completion}}{C++23 (ISO/IEC 14882:2023) - The Completion}}\label{c23-isoiec-148822023---the-completion}}

\textbf{Released:} October 2023 * \textbf{Language:} Deducing
\passthrough{\lstinline!this!} (Explicit object parameter),
\passthrough{\lstinline!if consteval!}, Multidimensional subscript
\passthrough{\lstinline!m[1,2]!}, Static
\passthrough{\lstinline!operator()!}, \passthrough{\lstinline!auto(x)!}
decay copy. * \textbf{Library:} \passthrough{\lstinline!std::print!},
\passthrough{\lstinline!std::println!},
\passthrough{\lstinline!std::expected!} (Error handling),
\passthrough{\lstinline!std::mdspan!},
\passthrough{\lstinline!std::flat\_map!},
\passthrough{\lstinline!std::flat\_set!},
\passthrough{\lstinline!std::generator!} (Synchronous coroutines),
\passthrough{\lstinline!std::stacktrace!},
\passthrough{\lstinline!std::stdatomic.h!}.
