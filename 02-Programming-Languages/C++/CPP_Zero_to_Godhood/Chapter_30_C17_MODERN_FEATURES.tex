\hypertarget{why-c17-matters}{%

C++17 addresses real pain points: - Safer null handling (optional) -
Type-safe unions (variant) - Zero-copy string operations (string\_view)
- Compile-time branching (if constexpr) - Pattern matching (structured
bindings) - Safe filesystem access - Automatic template deduction -
Flexible type storage (any)

\hypertarget{structured-bindings}{%
\section{STRUCTURED BINDINGS}\label{structured-bindings}}

\hypertarget{introduction-to-structured-bindings}{%
\section{1.1 Introduction to Structured
Bindings}\label{introduction-to-structured-bindings}}

Structured bindings allow unpacking objects into individual variables.

\hypertarget{basic-structured-bindings}{%
\subsection{Basic Structured Bindings}\label{basic-structured-bindings}}

\begin{lstlisting}[language={C++}]
#include <tuple>
#include <iostream>
using namespace std;

// Before C++17: Verbose
tuple<int, double, string> t = {42, 3.14, "hello"};
int a = get<0>(t);
double b = get<1>(t);
string c = get<2>(t);

// C++17: Clean and simple
auto [x, y, z] = t;
cout << x << ", " << y << ", " << z << "\n";  // 42, 3.14, hello
\end{lstlisting}

\hypertarget{structured-bindings-with-pairs}{%
\subsection{Structured Bindings with
Pairs}\label{structured-bindings-with-pairs}}

\begin{lstlisting}[language={C++}]
#include <map>

map<string, int> ages{{"Alice", 30}, {"Bob", 25}};

// Before C++17: Awkward
for (const auto& pair : ages) {
    const string& name = pair.first;
    int age = pair.second;
    cout << name << " is " << age << "\n";
}

// C++17: Natural
for (const auto& [name, age] : ages) {
    cout << name << " is " << age << "\n";
}
\end{lstlisting}

\hypertarget{structured-bindings-with-arrays}{%
\subsection{Structured Bindings with
Arrays}\label{structured-bindings-with-arrays}}

\begin{lstlisting}[language={C++}]
// Arrays
int arr[3] = {1, 2, 3};
auto [a, b, c] = arr;
cout << a << ", " << b << ", " << c << "\n";  // 1, 2, 3

// Fixed-size arrays
array<double, 4> coords = {1.0, 2.0, 3.0, 4.0};
auto [x, y, z, w] = coords;
\end{lstlisting}

\hypertarget{structured-bindings-with-structs}{%
\subsection{Structured Bindings with
Structs}\label{structured-bindings-with-structs}}

\begin{lstlisting}[language={C++}]
struct Person {
    string name;
    int age;
    string city;
};

Person p{"Alice", 30, "NYC"};

// Before C++17
string name = p.name;
int age = p.age;
string city = p.city;

// C++17
auto [name, age, city] = p;
cout << name << " is " << age << " from " << city << "\n";
\end{lstlisting}

\hypertarget{structured-bindings-with-return-values}{%
\subsection{Structured Bindings with Return
Values}\label{structured-bindings-with-return-values}}

\begin{lstlisting}[language={C++}]
pair<bool, int> divide(int a, int b) {
    if (b == 0) return {false, 0};
    return {true, a / b};
}

// Before C++17: Awkward
auto result = divide(10, 2);
if (result.first) {
    cout << "Result: " << result.second << "\n";
}

// C++17: Clear
auto [success, value] = divide(10, 2);
if (success) {
    cout << "Result: " << value << "\n";
}
\end{lstlisting}

\hypertarget{structured-bindings-with-references}{%
\subsection{Structured Bindings with
References}\label{structured-bindings-with-references}}

\begin{lstlisting}[language={C++}]
int x = 5, y = 10;

// Modify through references
auto& [rx, ry] = tuple<int&, int&>(x, y);
rx = 20;
cout << x << "\n";  // 20 (modified)

// Const references
const auto& [cx, cy] = tuple<int, int>(5, 10);
// cx = 100;  // ERROR - const
\end{lstlisting}

\hypertarget{practical-use-cases}{%
\subsection{Practical Use Cases}\label{practical-use-cases}}

\begin{lstlisting}[language={C++}]
// Function returning multiple values
tuple<bool, string, int> parse_config(const string& path) {
    // Parse and return success, name, port
    return {true, "server", 8080};
}

auto [ok, name, port] = parse_config("/etc/config");
if (ok) {
    cout << "Server " << name << " on port " << port << "\n";
}

// Iterating over map with unpacking
map<string, vector<int>> data{
    {"a", {1, 2, 3}},
    {"b", {4, 5, 6}}
};

for (auto [key, values] : data) {
    cout << key << ": ";
    for (int v : values) cout << v << " ";
    cout << "\n";
}

// Database-like access
vector<tuple<int, string, double>> records{
    {1, "Alice", 95.5},
    {2, "Bob", 87.0},
    {3, "Carol", 92.5}
};

for (auto [id, name, score] : records) {
    cout << id << ": " << name << " scored " << score << "\n";
}
\end{lstlisting}

\hypertarget{structured-bindings-rules}{%
\subsection{Structured Bindings Rules}\label{structured-bindings-rules}}

\begin{lstlisting}[language={C++}]
// Auto deduction (copies)
tuple<int, int> t{1, 2};
auto [a, b] = t;           // a, b are copies

// References
auto& [x, y] = t;          // x, y reference t's elements
const auto& [cx, cy] = t;  // const references

// Move
auto&& [mx, my] = move(t); // rvalue references

// Partial binding (with operator[])
struct Container {
    int& operator[](size_t i);  // Must return reference
};

Container c;
auto [elem] = c;           // Gets copy via operator[]

// Multiple items
auto [a, b, c, d] = tuple{1, 2, 3, 4};
auto [x, y, z] = array{10, 20, 30};
\end{lstlisting}

\hypertarget{structured-bindings-under-the-hood}{%
\section{1.2 Structured Bindings Under the
Hood}\label{structured-bindings-under-the-hood}}

The compiler generates a hidden variable.

\textbf{Code:}

\begin{lstlisting}[language={C++}]
auto [x, y] = my_pair;
\end{lstlisting}

\textbf{Compiler Logic:}

\begin{lstlisting}[language={C++}]
auto __hidden = my_pair;
auto& x = __hidden.first;  // Aliases
auto& y = __hidden.second;
\end{lstlisting}

\textbf{Implication}: * \passthrough{\lstinline!x!} and
\passthrough{\lstinline!y!} are NOT variables; they are names referring
to subobjects of the hidden variable. * If you use
\passthrough{\lstinline!auto\& [x, y]!}, the hidden variable is a
reference.

\hypertarget{optional-variant}{%
\section{OPTIONAL \& VARIANT}\label{optional-variant}}

\hypertarget{stdoptional---safe-nullable-values}{%
\section{2.1 std::optional - Safe Nullable
Values}\label{stdoptional---safe-nullable-values}}

\passthrough{\lstinline!std::optional!} represents a value that may or
may not be present.

\hypertarget{basic-optional-usage}{%
\subsection{Basic optional Usage}\label{basic-optional-usage}}

\begin{lstlisting}[language={C++}]
#include <optional>
#include <iostream>
using namespace std;

// Function that might not return a value
optional<int> parse_int(const string& s) {
    try {
        return stoi(s);
    } catch (...) {
        return nullopt;  // No value
    }
}

// Usage
auto result1 = parse_int("42");
if (result1.has_value()) {
    cout << "Value: " << result1.value() << "\n";
}

// Or using operator*
if (result1) {
    cout << "Value: " << *result1 << "\n";
}

// Default value
auto value = result1.value_or(0);  // 0 if no value
\end{lstlisting}

\hypertarget{optional-with-complex-types}{%
\subsection{optional with Complex
Types}\label{optional-with-complex-types}}

\begin{lstlisting}[language={C++}]
struct User {
    int id;
    string name;
};

optional<User> find_user(int id) {
    if (id < 0) return nullopt;
    return User{id, "User" + to_string(id)};
}

// Usage
if (auto user = find_user(42)) {
    cout << user->name << "\n";
} else {
    cout << "User not found\n";
}
\end{lstlisting}

\hypertarget{optional-operations}{%
\subsection{optional Operations}\label{optional-operations}}

\begin{lstlisting}[language={C++}]
optional<int> opt(42);

// Check
if (opt) cout << "Has value\n";          // true
if (opt.has_value()) cout << "Has\n";    // true

// Access
cout << opt.value() << "\n";              // 42
cout << *opt << "\n";                     // 42
cout << opt.value_or(0) << "\n";          // 42

// Modify
opt.value() = 100;
cout << *opt << "\n";                     // 100

// Reset
opt = nullopt;
if (opt) cout << "Has value\n";          // false

// Or assign
opt = 99;
cout << *opt << "\n";                     // 99
\end{lstlisting}

\hypertarget{optional-with-chaining}{%
\subsection{optional with Chaining}\label{optional-with-chaining}}

\begin{lstlisting}[language={C++}]
optional<int> process(optional<int> input) {
    if (!input) return nullopt;
    return input.value() * 2;
}

auto result = process(optional<int>(5));
if (result) {
    cout << result.value() << "\n";  // 10
}

// Chain operations
auto chain = parse_int("42")
    .and_then([](int x) -> optional<int> { return x * 2; })
    .or_else([]() { return optional<int>(0); });
\end{lstlisting}

\hypertarget{stdvariant---type-safe-union}{%
\section{2.2 std::variant - Type-Safe
Union}\label{stdvariant---type-safe-union}}

\passthrough{\lstinline!std::variant!} is a type-safe union that holds
one of several types.

\hypertarget{basic-variant-usage}{%
\subsection{Basic variant Usage}\label{basic-variant-usage}}

\begin{lstlisting}[language={C++}]
#include <variant>
#include <iostream>
using namespace std;

// Can hold int, double, or string
variant<int, double, string> value;

// Store int
value = 42;
cout << get<int>(value) << "\n";  // 42

// Store double
value = 3.14;
cout << get<double>(value) << "\n";  // 3.14

// Store string
value = string("hello");
cout << get<string>(value) << "\n";  // hello

// Check type
cout << value.index() << "\n";  // 2 (string is third)
\end{lstlisting}

\hypertarget{variant-with-type-checking}{%
\subsection{variant with Type
Checking}\label{variant-with-type-checking}}

\begin{lstlisting}[language={C++}]
void process(variant<int, double, string> value) {
    if (holds_alternative<int>(value)) {
        cout << "Integer: " << get<int>(value) << "\n";
    } else if (holds_alternative<double>(value)) {
        cout << "Double: " << get<double>(value) << "\n";
    } else if (holds_alternative<string>(value)) {
        cout << "String: " << get<string>(value) << "\n";
    }
}

process(42);              // "Integer: 42"
process(3.14);            // "Double: 3.14"
process("hello");         // "String: hello"
\end{lstlisting}

\hypertarget{variant-with-visitor-pattern}{%
\subsection{variant with Visitor
Pattern}\label{variant-with-visitor-pattern}}

\begin{lstlisting}[language={C++}]
struct Visitor {
    void operator()(int i) const {
        cout << "Integer: " << i << "\n";
    }
    
    void operator()(double d) const {
        cout << "Double: " << d << "\n";
    }
    
    void operator()(const string& s) const {
        cout << "String: " << s << "\n";
    }
};

variant<int, double, string> v = 42;
visit(Visitor(), v);  // "Integer: 42"

v = 3.14;
visit(Visitor(), v);  // "Double: 3.14"

v = string("hello");
visit(Visitor(), v);  // "String: hello"
\end{lstlisting}

\hypertarget{variant-with-lambdas-c20-or-using-overload-trick}{%
\subsection{variant with Lambdas (C++20 or using overload
trick)}\label{variant-with-lambdas-c20-or-using-overload-trick}}

\begin{lstlisting}[language={C++}]
// C++17 overload trick
template<typename... Ts>
struct overload : Ts... { using Ts::operator()...; };
template<typename... Ts>
overload(Ts...) -> overload<Ts...>;

variant<int, double, string> v = 42;

// Visit with lambdas
visit(overload{
    [](int i) { cout << "Integer: " << i << "\n"; },
    [](double d) { cout << "Double: " << d << "\n"; },
    [](const string& s) { cout << "String: " << s << "\n"; }
}, v);  // "Integer: 42"
\end{lstlisting}

\hypertarget{practical-variant-example}{%
\subsection{Practical variant Example}\label{practical-variant-example}}

\begin{lstlisting}[language={C++}]
variant<int, string> parse_value(const string& input) {
    try {
        return stoi(input);  // Try as int
    } catch (...) {
        return input;        // Return as string
    }
}

auto result = parse_value("42");
if (holds_alternative<int>(result)) {
    cout << "Parsed as int: " << get<int>(result) << "\n";
} else {
    cout << "Parsed as string: " << get<string>(result) << "\n";
}
\end{lstlisting}

\hypertarget{stdany}{%
\section{STD::ANY}\label{stdany}}

\hypertarget{type-erased-storage-with-stdany}{%
\section{3.1 Type-Erased Storage with
std::any}\label{type-erased-storage-with-stdany}}

\passthrough{\lstinline!std::any!} can hold any copyable type.

\hypertarget{basic-any-usage}{%
\subsection{Basic any Usage}\label{basic-any-usage}}

\begin{lstlisting}[language={C++}]
#include <any>
#include <iostream>
using namespace std;

any value;

// Store different types
value = 42;
cout << any_cast<int>(value) << "\n";  // 42

value = 3.14;
cout << any_cast<double>(value) << "\n";  // 3.14

value = string("hello");
cout << any_cast<string>(value) << "\n";  // hello

// Check type
if (value.type() == typeid(string)) {
    cout << "It's a string\n";
}
\end{lstlisting}

\hypertarget{any-with-type-checking}{%
\subsection{any with Type Checking}\label{any-with-type-checking}}

\begin{lstlisting}[language={C++}]
any value = 42;

if (value.type() == typeid(int)) {
    cout << "Value: " << any_cast<int>(value) << "\n";
}

// Safe cast (throws if wrong type)
try {
    double d = any_cast<double>(value);  // Wrong type
} catch (const bad_any_cast& e) {
    cout << "Type mismatch: " << e.what() << "\n";
}

// Check before casting
if (value.type() == typeid(int)) {
    int i = any_cast<int>(value);
}
\end{lstlisting}

\hypertarget{any-in-collections}{%
\subsection{any in Collections}\label{any-in-collections}}

\begin{lstlisting}[language={C++}]
vector<any> data;
data.push_back(42);
data.push_back(3.14);
data.push_back(string("hello"));
data.push_back(vector<int>{1, 2, 3});

// Process
for (auto& item : data) {
    if (item.type() == typeid(int)) {
        cout << "Int: " << any_cast<int>(item) << "\n";
    } else if (item.type() == typeid(double)) {
        cout << "Double: " << any_cast<double>(item) << "\n";
    } else if (item.type() == typeid(string)) {
        cout << "String: " << any_cast<string>(item) << "\n";
    }
}
\end{lstlisting}

\hypertarget{any-vs-variant}{%
\subsection{any vs variant}\label{any-vs-variant}}

\begin{lstlisting}[language={C++}]
// variant<int, double, string>: Fixed types, type-safe
variant<int, double, string> v = 42;
cout << get<int>(v) << "\n";  // Type-safe, no runtime check needed

// any: Any type, runtime type checking
any a = 42;
cout << any_cast<int>(a) << "\n";  // Runtime check, potential exception

// Use variant when types are known
// Use any when types are truly dynamic
\end{lstlisting}

\hypertarget{if-constexpr}{%
\section{IF CONSTEXPR}\label{if-constexpr}}

\hypertarget{compile-time-conditional-code}{%
\section{5.1 Compile-Time Conditional
Code}\label{compile-time-conditional-code}}

\passthrough{\lstinline!if constexpr!} allows branching at compile-time.

\hypertarget{basic-if-constexpr}{%
\subsection{Basic if constexpr}\label{basic-if-constexpr}}

\begin{lstlisting}[language={C++}]
#include <type_traits>

template<typename T>
void process(T value) {
    if constexpr (is_integral_v<T>) {
        cout << "Integer: " << value << "\n";
    } else if constexpr (is_floating_point_v<T>) {
        cout << "Float: " << value << "\n";
    } else {
        cout << "Other type\n";
    }
}

process(42);        // "Integer: 42" - int branch only compiled
process(3.14);      // "Float: 3.14" - double branch only compiled
process("hello");   // "Other type"
\end{lstlisting}

\hypertarget{if-constexpr-advantages}{%
\subsection{if constexpr Advantages}\label{if-constexpr-advantages}}

\begin{lstlisting}[language={C++}]
// Before C++17: SFINAE (complex, verbose)
template<typename T>
enable_if_t<is_integral_v<T>>
old_way(T x) { cout << "Int\n"; }

template<typename T>
enable_if_t<is_floating_point_v<T>>
old_way(T x) { cout << "Float\n"; }

// After C++17: if constexpr (clean, readable)
template<typename T>
void new_way(T x) {
    if constexpr (is_integral_v<T>) {
        cout << "Int\n";
    } else if constexpr (is_floating_point_v<T>) {
        cout << "Float\n";
    }
}
\end{lstlisting}

\hypertarget{if-constexpr-with-complex-logic}{%
\subsection{if constexpr with Complex
Logic}\label{if-constexpr-with-complex-logic}}

\begin{lstlisting}[language={C++}]
template<typename T>
void serialize(const T& value) {
    if constexpr (is_arithmetic_v<T>) {
        // Serialize numbers efficiently
        write_binary(value);
    } else if constexpr (is_same_v<T, string>) {
        // Serialize strings
        write_string(value);
    } else if constexpr (requires { value.size(); }) {
        // Serialize containers
        for (const auto& item : value) {
            serialize(item);
        }
    }
}
\end{lstlisting}

\hypertarget{if-constexpr-with-concepts-c20-like}{%
\subsection{if constexpr with Concepts
(C++20-like)}\label{if-constexpr-with-concepts-c20-like}}

\begin{lstlisting}[language={C++}]
template<typename T>
void print_info(const T& value) {
    cout << "Type: " << typeid(T).name() << "\n";
    
    if constexpr (requires { value.size(); }) {
        cout << "Size: " << value.size() << "\n";
    }
    
    if constexpr (requires { value.data(); }) {
        cout << "Data pointer available\n";
    }
    
    if constexpr (is_default_constructible_v<T>) {
        T temp;  // Can create temporary
    }

\end{lstlisting}

\hypertarget{the-death-of-sfinae}{%
\section{5.2 The Death of SFINAE?}\label{the-death-of-sfinae}}

\passthrough{\lstinline!if constexpr!} replaces SFINAE for
\emph{implementation details}, but not for \emph{overload resolution}.

\textbf{SFINAE (Old Way - C++11/14):}

\begin{lstlisting}[language={C++}]
template <typename T>
typename std::enable_if<std::is_integral<T>::value, T>::type
gcd(T a, T b) { return b == 0 ? a : gcd(b, a % b); }

template <typename T>
typename std::enable_if<!std::is_integral<T>::value, T>::type
gcd(T a, T b) { static_assert(std::is_integral<T>::value, "GCD not for floats"); }
\end{lstlisting}

\textbf{if constexpr (New Way - C++17):}

\begin{lstlisting}[language={C++}]
template <typename T>
T gcd(T a, T b) {
    if constexpr (std::is_integral_v<T>) {
        return b == 0 ? a : gcd(b, a % b);
    } else {
        static_assert(always_false<T>, "GCD not for floats");
    }
}
\end{lstlisting}

\emph{Result}: Much cleaner, easier to debug errors.

\hypertarget{fold-expressions}{%
\section{FOLD EXPRESSIONS}\label{fold-expressions}}

\hypertarget{operating-on-parameter-packs}{%
\section{6.1 Operating on Parameter
Packs}\label{operating-on-parameter-packs}}

Fold expressions simplify operations on variadic templates.

\hypertarget{basic-fold-expressions}{%
\subsection{Basic Fold Expressions}\label{basic-fold-expressions}}

\begin{lstlisting}[language={C++}]
#include <iostream>
using namespace std;

// Sum with fold
template<typename... Args>
int sum(Args... args) {
    return (... + args);  // Left fold: ((a + b) + c) + d
}

cout << sum(1, 2, 3, 4) << "\n";  // 10

// Product with fold
template<typename... Args>
int product(Args... args) {
    return (... * args);  // Left fold: ((a * b) * c) * d
}

cout << product(2, 3, 4) << "\n";  // 24

// Logical AND
template<typename... Args>
bool all_true(Args... args) {
    return (... && args);
}

cout << all_true(true, true, true) << "\n";      // true
cout << all_true(true, false, true) << "\n";     // false
\end{lstlisting}

\hypertarget{fold-directions}{%
\subsection{Fold Directions}\label{fold-directions}}

\begin{lstlisting}[language={C++}]
// Left fold: (... + args) = ((a + b) + c) + d
template<typename... Args>
int left_fold(Args... args) {
    return (... + args);
}

// Right fold: (args + ...) = a + (b + (c + d))
template<typename... Args>
int right_fold(Args... args) {
    return (args + ...);
}

// For addition, result is the same
left_fold(1, 2, 3);   // ((1 + 2) + 3) = 6
right_fold(1, 2, 3);  // (1 + (2 + 3)) = 6

// For subtraction, different!
// left: ((1 - 2) - 3) = -4
// right: (1 - (2 - 3)) = 2
\end{lstlisting}

\hypertarget{fold-with-default-value}{%
\subsection{Fold with Default Value}\label{fold-with-default-value}}

\begin{lstlisting}[language={C++}]
// No default value
template<typename... Args>
int sum_no_default(Args... args) {
    return (... + args);  // Error if no args
}

// With default value
template<typename... Args>
int sum_default(Args... args) {
    return (args + ... + 0);  // 0 if no args
}

sum_default();              // 0
sum_default(1, 2, 3);       // 6
\end{lstlisting}

\hypertarget{practical-fold-examples}{%
\subsection{Practical Fold Examples}\label{practical-fold-examples}}

\begin{lstlisting}[language={C++}]
// Print all arguments
template<typename... Args>
void print_all(Args... args) {
    ((cout << args << " "), ...);
}

print_all(1, "hello", 3.14, "world");
// Output: 1 hello 3.14 world

// Check if any true
template<typename... Args>
bool any_true(Args... args) {
    return (... || args);
}

any_true(false, false, true, false);  // true

// Maximum of values
template<typename... Args>
int maximum(Args... args) {
    return max({args...});  // Fold into initializer
}

cout << maximum(3, 1, 4, 1, 5, 9) << "\n";  // 9
\end{lstlisting}

\hypertarget{automatic-template-type-deduction}{%
\section{7.1 Automatic Template Type
Deduction}\label{automatic-template-type-deduction}}

CTAD allows deduction of class template parameters from constructor
arguments.

\hypertarget{before-ctad-c14}{%
\subsection{Before CTAD (C++14)}\label{before-ctad-c14}}

\begin{lstlisting}[language={C++}]

#include <map>
using namespace std;

// Must specify types explicitly

\end{lstlisting}

\hypertarget{with-ctad-c17}{%
\subsection{With CTAD (C++17)}\label{with-ctad-c17}}

\begin{lstlisting}[language={C++}]
// Types deduced automatically!
vector v{1, 2, 3};              // Deduced as vector<int>
pair p("hello", 42);             // Deduced as pair<string, int>
map m{pair("a", 1), pair("b", 2)}; // Deduced as map<string, int>

// Works with custom classes too
template<typename T, typename U>
struct Pair {
    T first;
    U second;
};

Pair p{42, 3.14};  // Deduced as Pair<int, double>
\end{lstlisting}

\hypertarget{ctad-with-custom-deduction-guides}{%
\subsection{CTAD with Custom Deduction
Guides}\label{ctad-with-custom-deduction-guides}}

\begin{lstlisting}[language={C++}]
template<typename T>
struct Container {
    vector<T> data;
};

// Without guide: Would fail
// Container c{1, 2, 3};  // ERROR - can't deduce T

// With deduction guide
template<typename T>
Container(initializer_list<T>) -> Container<T>;

// Now it works!
Container c{1, 2, 3};  // Deduced as Container<int>
\end{lstlisting}

    U second;

\hypertarget{introduction-to-stdpmr}{%

C++17 introduces \passthrough{\lstinline!std::pmr!} (Polymorphic Memory
Resources) in \passthrough{\lstinline!<memory\_resource>!}, enabling
efficient, customizable memory management without changing container
types.

\hypertarget{the-problem-with-traditional-allocators}{%
\subsection{The Problem with Traditional
Allocators}\label{the-problem-with-traditional-allocators}}

Traditional allocators are part of the type signature:
\passthrough{\lstinline!std::vector<int, MyAlloc<int>>!} is a different
type from \passthrough{\lstinline!std::vector<int>!}.

\passthrough{\lstinline!std::pmr!} erases the allocator type, allowing
containers with different allocation strategies to be used
interchangeably.

\begin{lstlisting}[language={C++}]
#include <vector>
#include <memory_resource>
#include <iostream>

// Use pmr namespace
namespace pmr = std::pmr;

void process_data(const pmr::vector<int>& data) {
    // Works with ANY allocator (stack, heap, pool)
    for (int x : data) std::cout << x << " ";
}

int main() {
    // 1. Default allocator (Heap)
    pmr::vector<int> heap_vec = {1, 2, 3};
    process_data(heap_vec);
    
    // 2. Stack allocator (Monotonic Buffer)
    std::array<std::byte, 1024> buffer;
    pmr::monotonic_buffer_resource pool{
        buffer.data(), buffer.size(), pmr::null_memory_resource()
    };
    
    pmr::vector<int> stack_vec(&pool);
    stack_vec.push_back(4);
    stack_vec.push_back(5);
    process_data(stack_vec);
    
    return 0;
}
\end{lstlisting}

\hypertarget{standard-memory-resources}{%
\subsection{Standard Memory Resources}\label{standard-memory-resources}}

\begin{lstlisting}[language={C++}]
#include <memory_resource>

// 1. new_delete_resource (Global Heap)
auto* heap = std::pmr::new_delete_resource();

// 2. null_memory_resource (Throws bad_alloc)
auto* null = std::pmr::null_memory_resource();

// 3. monotonic_buffer_resource (Fast, no deallocation)
// Very fast for building complex structures, deallocates all at once
std::pmr::monotonic_buffer_resource fast_pool(heap);

// 4. synchronized_pool_resource (Thread-safe pool)
// Good for many small allocations of same size
std::pmr::synchronized_pool_resource thread_safe_pool(heap);

// 5. unsynchronized_pool_resource (Single-thread pool)
// Fastest for single-threaded small allocations
std::pmr::unsynchronized_pool_resource local_pool(heap);
\end{lstlisting}

\hypertarget{chaining-resources}{%
\subsection{Chaining Resources}\label{chaining-resources}}

Memory resources can be chained. If a pool runs out of memory, it
requests more from an ``upstream'' resource.

\begin{lstlisting}[language={C++}]
#include <memory_resource>
#include <vector>

void chaining_example() {
    // Buffer on stack
    std::array<std::byte, 256> buffer;
    
    // Primary: Use stack buffer
    // Upstream: If buffer full, go to heap
    std::pmr::monotonic_buffer_resource mem_res(
        buffer.data(), buffer.size(), std::pmr::new_delete_resource()
    );
    
    std::pmr::vector<int> vec(&mem_res);
    
    // These go to stack buffer
    for (int i = 0; i < 50; i++) vec.push_back(i);
    
    // If we exceed buffer, it silently falls back to heap
}
\end{lstlisting}

\subsection{Performance Benefits}\label{performance-benefits}}

\passthrough{\lstinline!std::pmr!} allows easy implementation of
\textbf{Arena Allocation} or \textbf{Stack Allocation} for standard
containers, which can provide massive performance gains (cache locality,
no malloc overhead) for short-lived complex data structures.
