\hypertarget{concepts-utilities}{%
\subsection{Concepts \& Utilities}\label{concepts-utilities}}

\begin{itemize}
\tightlist
\item
  \passthrough{\lstinline!<concepts>!} (C++20): Fundamental concepts
  library.
\item
  \passthrough{\lstinline!<coroutine>!} (C++20): Coroutine support
  library.
\item
  \passthrough{\lstinline!<functional>!}: Function objects, binder, and
  reference wrappers.
\item
  \passthrough{\lstinline!<memory>!}: Smart pointers and allocators.
\item
  \passthrough{\lstinline!<tuple>!}: Tuple library.
\item
  \passthrough{\lstinline!<type\_traits>!}: Compile-time type
  information.
\item
  \passthrough{\lstinline!<utility>!}: Utility components
  (\passthrough{\lstinline!std::pair!},
  \passthrough{\lstinline!std::move!}).
\end{itemize}

\hypertarget{containers}{%
\subsection{Containers}\label{containers}}

\begin{itemize}
\tightlist
\item
  \passthrough{\lstinline!<array>!} (C++11): Fixed-size array class.
\item
  \passthrough{\lstinline!<deque>!}: Double-ended queue.
\item
  \passthrough{\lstinline!<list>!}: Doubly-linked list.
\item
  \passthrough{\lstinline!<map>!}: Associative containers (Red-Black
  Tree).
\item
  \passthrough{\lstinline!<queue>!}: Queue adapter.
\item
  \passthrough{\lstinline!<set>!}: Associative containers (Red-Black
  Tree).
\item
  \passthrough{\lstinline!<stack>!}: Stack adapter.
\item
  \passthrough{\lstinline!<unordered\_map>!} (C++11): Hash map.
\item
  \passthrough{\lstinline!<vector>!}: Dynamic array.
\item
  \passthrough{\lstinline!<span >!} (C++20): Non-owning view of
  contiguous memory.
\end{itemize}
