\hypertarget{stdfilesystem}{%
\subsection{\texorpdfstring{1.
\texttt{std::filesystem}}{1. std::filesystem}}\label{stdfilesystem}}

Standardized file system operations. Based on
\passthrough{\lstinline!boost::filesystem!}.

\hypertarget{paths}{%
\subsubsection{1.1 Paths}\label{paths}}

\begin{lstlisting}[language={C++}]
#include <filesystem>
namespace fs = std::filesystem;

fs::path p = "/home/user/data.txt";

std::cout << p.filename();      // "data.txt"
std::cout << p.extension();     // ".txt"
std::cout << p.parent_path();   // "/home/user"
\end{lstlisting}

\hypertarget{iterating-directories}{%
\subsubsection{1.2 Iterating Directories}\label{iterating-directories}}

\begin{lstlisting}[language={C++}]
for (const auto& entry : fs::directory_iterator("/home/user")) {
    std::cout << entry.path() << "\n";
}

// Recursive
for (const auto& entry : fs::recursive_directory_iterator("/home/user")) {
    // ...
}
\end{lstlisting}

\hypertarget{operations}{%
\subsubsection{1.3 Operations}\label{operations}}

\begin{lstlisting}[language={C++}]
fs::create_directory("sandbox");
fs::copy("a.txt", "b.txt");
fs::rename("b.txt", "c.txt");
fs::remove("c.txt"); // Returns true if removed
bool exists = fs::exists("sandbox");
uintmax_t size = fs::file_size("a.txt");
\end{lstlisting}

\hypertarget{polymorphic-allocators-stdpmr}{%
\subsection{\texorpdfstring{2. Polymorphic Allocators
(\texttt{std::pmr})}{2. Polymorphic Allocators (std::pmr)}}\label{polymorphic-allocators-stdpmr}}

Memory resource management that is detached from the type.

\begin{lstlisting}[language={C++}]
#include <memory_resource>
#include <vector>

char buffer[1024];
std::pmr::monotonic_buffer_resource pool(buffer, 1024);
std::pmr::vector<int> v(&pool); // Uses stack buffer!

v.push_back(1);
// No heap allocation happens until buffer is exhausted.
\end{lstlisting}
