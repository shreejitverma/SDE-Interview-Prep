\hypertarget{stdmdspan}{%
\section{\texorpdfstring{1.
\texttt{std::mdspan}}{1. std::mdspan}}\label{stdmdspan}}

A non-owning, multidimensional view of a contiguous memory block.
Crucial for scientific computing, linear algebra, and image processing.

\hypertarget{basics}{%
\subsection{1.1 Basics}\label{basics}}

\begin{lstlisting}[language={C++}]
#include <mdspan>
#include <vector>

int main() {
    std::vector<int> data(12); // 3x4 matrix
    std::mdspan m(data.data(), 3, 4);
    
    m[1, 2] = 42; // Set row 1, col 2
}
\end{lstlisting}

\hypertarget{layouts}{%
\subsection{1.2 Layouts}\label{layouts}}

\begin{itemize}
\tightlist
\item
  \passthrough{\lstinline!std::layout\_right!} (Row-Major): C/C++
  default. Last index varies fastest.
\item
  \passthrough{\lstinline!std::layout\_left!} (Column-Major):
  Fortran/BLAS compatible. First index varies fastest.
\end{itemize}

\hypertarget{stdflat_map-and-stdflat_set}{%
\section{\texorpdfstring{2. \texttt{std::flat\_map} and
\texttt{std::flat\_set}}{2. std::flat\_map and std::flat\_set}}\label{stdflat_map-and-stdflat_set}}

Associative containers implemented as sorted vectors.

\begin{itemize}
\tightlist
\item
  \textbf{Pros:} contiguous memory, cache-friendly, faster
  iteration/lookup for small-to-medium datasets.
\item
  \textbf{Cons:} O(N) insertion/deletion (vs O(log N) for
  \passthrough{\lstinline!std::map!}). Iterators invalidated on
  insertion.
\end{itemize}

\begin{lstlisting}[language={C++}]
#include <flat_map>

std::flat_map<int, std::string> m;
m[1] = "One"; // Insert
\end{lstlisting}

\hypertarget{ranges-enhancements}{%
\section{3. Ranges Enhancements}\label{ranges-enhancements}}

\begin{itemize}
\tightlist
\item
  \passthrough{\lstinline!std::ranges::to!}: Convert a range to a
  container.
  \passthrough{\lstinline!cpp     auto v = some\_view | std::ranges::to<std::vector>();!}
\item
  \passthrough{\lstinline!views::zip!}: Iterate multiple ranges in
  lockstep.
  \passthrough{\lstinline!cpp     std::vector<int> a = \{1, 2\}, b = \{3, 4\};     for (auto [x, y] : std::views::zip(a, b)) \{         // x=1, y=3 ...     \}!}
\item
  \passthrough{\lstinline!views::enumerate!}: Index + Value.
  \passthrough{\lstinline!cpp     for (auto [idx, val] : std::views::enumerate(data)) \{         // ...     \}!}
\end{itemize}
