\hypertarget{stdmake_unique}{%
\section{\texorpdfstring{1.
\texttt{std::make\_unique}}{1. std::make\_unique}}\label{stdmake_unique}}

The missing counterpart to \passthrough{\lstinline!std::make\_shared!}.

\begin{lstlisting}[language={C++}]
auto ptr = std::make_unique<int>(42);
// Exception safe, no `new` keyword.
\end{lstlisting}

\hypertarget{shared-locks-stdshared_timed_mutex}{%
\section{\texorpdfstring{2. Shared Locks
(\texttt{std::shared\_timed\_mutex})}{2. Shared Locks (std::shared\_timed\_mutex)}}\label{shared-locks-stdshared_timed_mutex}}

Reader-writer locking. Multiple readers can hold a shared lock; writers
need an exclusive lock.

\begin{lstlisting}[language={C++}]
#include <shared_mutex>

std::shared_timed_mutex mtx;

void reader() {
    std::shared_lock<std::shared_timed_mutex> lock(mtx);
    // read
}

void writer() {
    std::unique_lock<std::shared_timed_mutex> lock(mtx);
    // write
}
\end{lstlisting}

\hypertarget{stdexchange}{%
\section{\texorpdfstring{3.
\texttt{std::exchange}}{3. std::exchange}}\label{stdexchange}}

Assigns a new value to an object and returns the old value.

\begin{lstlisting}[language={C++}]
int x = 10;
int old = std::exchange(x, 20); // x=20, old=10
\end{lstlisting}

\hypertarget{stdquoted}{%
\section{\texorpdfstring{4.
\texttt{std::quoted}}{4. std::quoted}}\label{stdquoted}}

Quoted string I/O for streams.

\begin{lstlisting}[language={C++}]
#include <iomanip>
std::cout << std::quoted("Hello World"); // "Hello World"
\end{lstlisting}

Access tuple elements by type (if unique).

int i = std::get<int>(t);
