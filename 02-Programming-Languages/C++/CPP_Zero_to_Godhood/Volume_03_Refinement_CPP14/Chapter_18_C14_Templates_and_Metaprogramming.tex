\hypertarget{variable-templates}{%
\section{1. Variable Templates}\label{variable-templates}}

Templates for variables, not just functions or classes.

\begin{lstlisting}[language={C++}]
template<typename T>
constexpr T pi = T(3.1415926535897932385);

int main() {
    float f = pi<float>;
    double d = pi<double>;
}
\end{lstlisting}

\hypertarget{decltypeauto}{%
\section{\texorpdfstring{2.
\texttt{decltype(auto)}}{2. decltype(auto)}}\label{decltypeauto}}

Deduces type exactly as \passthrough{\lstinline!decltype!} would, but
without repeating the expression.

\begin{itemize}
\tightlist
\item
  \passthrough{\lstinline!auto!}: Deduces type (decaying references).
\item
  \passthrough{\lstinline!decltype(auto)!}: Deduces type and value
  category (preserves references).
\end{itemize}

\begin{lstlisting}[language={C++}]
int x = 5;
int& ref = x;

auto a = ref;           // int
decltype(auto) b = ref; // int&
\end{lstlisting}

\hypertarget{standard-library-metafunctions}{%
\section{3. Standard Library
Metafunctions}\label{standard-library-metafunctions}}

\begin{itemize}
\tightlist
\item
  \passthrough{\lstinline!std::integer\_sequence!}
\item
  \passthrough{\lstinline!std::index\_sequence!}
\item
  \passthrough{\lstinline!std::make\_index\_sequence!}
\end{itemize}

Useful for unpacking tuples or variadic templates.

\begin{lstlisting}[language={C++}]
template<typename Tuple, size_t... Is>
void print_tuple(const Tuple& t, std::index_sequence<Is...>) {
    ((std::cout << std::get<Is>(t) << " "), ...);
}
\end{lstlisting}
