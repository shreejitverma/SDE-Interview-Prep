\chapter{Appendix B: Common Acronyms}\label{appendix-b-common-acronyms}}

\begin{itemize}
\tightlist
\item
  \textbf{ABI}: Application Binary Interface.
\item
  \textbf{API}: Application Programming Interface.
\item
  \textbf{COW}: Copy On Write.
\item
  \textbf{CRTP}: Curiously Recurring Template Pattern.
\item
  \textbf{CTAD}: Class Template Argument Deduction.
\item
  \textbf{UB}: Undefined Behavior (Avoid at all costs!).
\item
  \textbf{IB}: Implementation-defined Behavior.
\item
  \textbf{IIFE}: Immediately Invoked Function Expression (often with
  lambdas).
\item
  \textbf{NrvO / RVO}: (Named) Return Value Optimization.
\item
  \textbf{ODR}: One Definition Rule.
\item
  \textbf{PIMPL}: Pointer to Implementation (Opaque Pointer).
\item
  \textbf{RAII}: Resource Acquisition Is Initialization.
\item
  \textbf{RTTI}: Run-Time Type Information.
\item
  \textbf{SFINAE}: Substitution Failure Is Not An Error.
\item
  \textbf{SOO / SSO}: Small Object/String Optimization.
\item
  \textbf{STL}: Standard Template Library.
\item
  \textbf{TMP}: Template Metaprogramming.
\item
  \textbf{TU}: Translation Unit.
\end{itemize}
