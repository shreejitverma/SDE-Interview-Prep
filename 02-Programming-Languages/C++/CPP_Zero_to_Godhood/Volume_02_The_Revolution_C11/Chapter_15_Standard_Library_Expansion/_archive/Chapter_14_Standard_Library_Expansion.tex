\hypertarget{unordered-containers}{%
\section{1. Unordered Containers}\label{unordered-containers}}

Hash maps/sets. O(1) average access.

\begin{itemize}
\tightlist
\item
  \passthrough{\lstinline!std::unordered\_map!}
\item
  \passthrough{\lstinline!std::unordered\_set!}
\item
  \passthrough{\lstinline!std::unordered\_multimap!}
\item
  \passthrough{\lstinline!std::unordered\_multiset!}
\end{itemize}

Requires a hash function for the key type.

\hypertarget{stdarray}{%
\section{\texorpdfstring{2.
\texttt{std::array}}{2. std::array}}\label{stdarray}}

Fixed-size, stack-allocated array. Wrapper around C-style array with STL
interface.

\begin{lstlisting}[language={C++}]
std::array<int, 5> arr = {1, 2, 3, 4, 5};
// Bounds checking available: arr.at(10) throws
// Knows its size: arr.size()
// Doesn't decay to pointer automatically
\end{lstlisting}

\hypertarget{stdregex}{%
\section{\texorpdfstring{3.

#include <regex>
std::regex r("(\w+)@(\w+)\.com");
\end{lstlisting}

\hypertarget{stdchrono}{%
\section{\texorpdfstring{4.
\texttt{std::chrono}}{4. std::chrono}}\label{stdchrono}}

Type-safe time library.

\begin{itemize}
\tightlist
\item
  \textbf{Clocks:} \passthrough{\lstinline!system\_clock!},
  \passthrough{\lstinline!steady\_clock!}.
\item
  \textbf{Durations:} \passthrough{\lstinline!milliseconds!},
  \passthrough{\lstinline!seconds!}.
\item
  \textbf{Time Points:} \passthrough{\lstinline!time\_point!}.
\end{itemize}

\begin{lstlisting}[language={C++}]
auto start = std::chrono::steady_clock::now();
// ...
auto end = std::chrono::steady_clock::now();
auto diff = end - start;
\end{lstlisting}

\hypertarget{stdrandom}{%
\section{\texorpdfstring{5.
\texttt{std::random}}{5. std::random}}\label{stdrandom}}

Better random number generation than \passthrough{\lstinline!rand()!}.

\begin{lstlisting}[language={C++}]
#include <random>

std::random_device rd;
std::mt19937 gen(rd()); // Mersenne Twister
std::uniform_int_distribution<> dis(1, 6);

int roll = dis(gen);
\end{lstlisting}
