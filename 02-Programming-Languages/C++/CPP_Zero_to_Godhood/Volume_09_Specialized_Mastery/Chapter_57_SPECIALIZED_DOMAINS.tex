\hypertarget{automotive-misra-c-autosar}{%
\subsection{37.4 Automotive (MISRA C++ /
AUTOSAR)}\label{automotive-misra-c-autosar}}

\begin{itemize}
\tightlist
\item
  \textbf{No Dynamic Memory:} All containers have fixed capacity
  (\passthrough{\lstinline!etl::vector!} or
  \passthrough{\lstinline!boost::static\_vector!}).
\item
  \textbf{No Exceptions:} \passthrough{\lstinline!try-catch!} adds
  binary size and non-deterministic unwind paths. Use
  \passthrough{\lstinline!std::expected!} or error codes.
\item
  \textbf{Static Analysis:} Code must pass MISRA-2008 rules (e.g.,
  ``Rule 5-0-15: Array indexing shall be checked'').
\end{itemize}

\begin{lstlisting}[language={C++}]
// Stack-only pattern (Automotive safe)
template <typename T, size_t N>
class FixedVector {
    T data[N];
    size_t count = 0;
public:
    void push_back(const T& val) {
        if (count < N) data[count++] = val;
    }
};
\end{lstlisting}
