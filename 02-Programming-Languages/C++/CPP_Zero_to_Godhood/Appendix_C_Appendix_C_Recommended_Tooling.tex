\hypertarget{compilers}{%
\subsection{Compilers}\label{compilers}}

\begin{itemize}
\tightlist
\item
  \textbf{GCC (GNU Compiler Collection)}: Standard on Linux.
\item
  \textbf{Clang/LLVM}: Excellent error messages, widely used on
  macOS/Linux.
\item
  \textbf{MSVC (Microsoft Visual C++)}: Standard on Windows.
\end{itemize}

\hypertarget{build-systems}{%
\subsection{Build Systems}\label{build-systems}}

\begin{itemize}
\tightlist
\item
  \textbf{CMake}: The industry standard meta-build system.
\item
  \textbf{Meson}: Modern, fast, Python-based.
\item
  \textbf{Bazel}: Google's build system, good for monorepos.
\end{itemize}

\hypertarget{package-managers}{%
\subsection{Package Managers}\label{package-managers}}

\begin{itemize}
\tightlist
\item
  \textbf{Conan}: Decentralized package manager for C/C++.
\item
  \textbf{vcpkg}: Microsoft's C++ library manager.
\end{itemize}

\hypertarget{static-analysis-sanitizers}{%
\subsection{Static Analysis \&
Sanitizers}\label{static-analysis-sanitizers}}

\begin{itemize}
\tightlist
\item
  \textbf{AddressSanitizer (ASan)}: Detects memory errors (buffer
  overflows, use-after-free).
\item
  \textbf{UndefinedBehaviorSanitizer (UBSan)}: Detects undefined
  behavior.
\item
  \textbf{ThreadSanitizer (TSan)}: Detects data races.
\item
  \textbf{Clang-Tidy}: Linter and static analysis tool.
\item
  \textbf{Cppcheck}: Static analysis tool.
\end{itemize}
